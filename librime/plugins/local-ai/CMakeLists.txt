cmake_minimum_required(VERSION 3.12)
project(rime-local-ai)
if(NOT WIN32)
  message(FATAL_ERROR "local-ai currently uses Windows WinHTTP")
endif()
add_library(rime-local-ai-objs OBJECT local_ai.cc)
target_compile_definitions(rime-local-ai-objs PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set(plugin_name rime-local-ai PARENT_SCOPE)
set(plugin_objs $<TARGET_OBJECTS:rime-local-ai-objs> PARENT_SCOPE)
set(plugin_deps winhttp PARENT_SCOPE)
set(plugin_modules local_ai PARENT_SCOPE)
add_executable(local_ai_probe probe.cc)
target_link_libraries(local_ai_probe PRIVATE rime)
set_target_properties(local_ai_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
